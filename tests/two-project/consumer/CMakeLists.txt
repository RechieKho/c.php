cmake_minimum_required(VERSION 3.20)
project(consumer LANGUAGES C)
include(FetchContent)
set(CPHP_ROOT "" CACHE PATH "c.php checkout")
set(PRODUCER_INSTALL "" CACHE PATH "installed producer prefix")
if(PRODUCER_INSTALL)
  find_package(cphp REQUIRED)
  find_package(producer CONFIG REQUIRED)
else()
  FetchContent_Declare(cphp SOURCE_DIR "${CPHP_ROOT}")
  FetchContent_Declare(producer SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../producer")
  FetchContent_MakeAvailable(cphp producer)
endif()
add_executable(app main.c)
if(PRODUCER_INSTALL)
  target_link_libraries(app PRIVATE producer::producer)
else()
  target_link_libraries(app PRIVATE producer)
endif()
cphp_use_composer(app)
cphp_add_sources(app gen.c.php)
enable_testing()
add_test(NAME app COMMAND app)
