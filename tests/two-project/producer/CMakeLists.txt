cmake_minimum_required(VERSION 3.20)
project(producer LANGUAGES C)
if(NOT COMMAND cphp_export_php)
  find_package(cphp REQUIRED)
endif()
add_library(producer STATIC src/answer.c)
target_include_directories(producer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
cphp_export_php(producer DIRECTORY php NAME producer)
cphp_use_composer(producer)
install(TARGETS producer EXPORT producerTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT producerTargets NAMESPACE producer:: DESTINATION lib/cmake/producer)
install(FILES producer-config.cmake DESTINATION lib/cmake/producer)
